-- AHNUM TASK: COMPLETE AUTOMATIC PROFILE AND REFERRAL SYSTEM -- 1. Ensure required columns exist ALTER TABLE public.profiles ADD COLUMN IF NOT EXISTS full_name text; ALTER TABLE public.profiles ADD COLUMN IF NOT EXISTS referral_code text; ALTER TABLE public.profiles ADD COLUMN IF NOT EXISTS referred_by uuid REFERENCES public.profiles(id); -- 2. Generate unique invitation codes CREATE OR REPLACE FUNCTION public.generate_referral_code() RETURNS text LANGUAGE plpgsql AS $$ DECLARE new_code text; BEGIN LOOP new_code := 'AHN' || upper(substr(replace(gen_random_uuid()::text, '-', ''), 1, 8)); EXIT WHEN NOT EXISTS ( SELECT 1 FROM public.profiles WHERE lower(referral_code) = lower(new_code) ); END LOOP; RETURN new_code; END; $$; -- 3. Automatically handle referral information CREATE OR REPLACE FUNCTION public.handle_profile_referral() RETURNS trigger LANGUAGE plpgsql SECURITY DEFINER SET search_path = public, auth AS $$ DECLARE incoming_code text; inviter_id uuid; BEGIN IF NEW.referral_code IS NULL OR trim(NEW.referral_code) = '' THEN NEW.referral_code := public.generate_referral_code(); END IF; IF NEW.referred_by IS NULL THEN SELECT raw_user_meta_data->>'referral_code' INTO incoming_code FROM auth.users WHERE id = NEW.id; IF incoming_code IS NOT NULL AND trim(incoming_code) <> '' THEN SELECT id INTO inviter_id FROM public.profiles WHERE lower(referral_code) = lower(trim(incoming_code)) AND id <> NEW.id LIMIT 1; NEW.referred_by := inviter_id; END IF; END IF; RETURN NEW; END; $$; -- 4. Install profile trigger if missing DO $$ BEGIN IF NOT EXISTS ( SELECT 1 FROM pg_trigger WHERE tgname = 'profile_referral_trigger' AND tgrelid = 'public.profiles'::regclass AND NOT tgisinternal ) THEN CREATE TRIGGER profile_referral_trigger BEFORE INSERT OR UPDATE ON public.profiles FOR EACH ROW EXECUTE FUNCTION public.handle_profile_referral(); END IF; END; $$; -- 5. Automatically create profiles after registration CREATE OR REPLACE FUNCTION public.handle_new_ahnum_user() RETURNS trigger LANGUAGE plpgsql SECURITY DEFINER SET search_path = public, auth AS $$ BEGIN INSERT INTO public.profiles (id, full_name) VALUES ( NEW.id, NULLIF(trim(NEW.raw_user_meta_data->>'full_name'), '') ) ON CONFLICT (id) DO UPDATE SET full_name = COALESCE( NULLIF(trim(EXCLUDED.full_name), ''), public.profiles.full_name ); RETURN NEW; END; $$; -- 6. Install automatic registration trigger DO $$ BEGIN IF NOT EXISTS ( SELECT 1 FROM pg_trigger WHERE tgname = 'ahnum_create_profile_after_signup' AND tgrelid = 'auth.users'::regclass AND NOT tgisinternal ) THEN CREATE TRIGGER ahnum_create_profile_after_signup AFTER INSERT ON auth.users FOR EACH ROW EXECUTE FUNCTION public.handle_new_ahnum_user(); END IF; END; $$; -- 7. Ensure referral codes are unique CREATE UNIQUE INDEX IF NOT EXISTS profiles_referral_code_unique ON public.profiles (lower(referral_code)) WHERE referral_code IS NOT NULL; -- 8. Repair existing members' names UPDATE public.profiles p SET full_name = NULLIF( trim(u.raw_user_meta_data->>'full_name'), '' ) FROM auth.users u WHERE p.id = u.id AND (p.full_name IS NULL OR trim(p.full_name) = ''); -- 9. Generate missing codes for existing members UPDATE public.profiles SET referral_code = public.generate_referral_code() WHERE referral_code IS NULL OR trim(referral_code) = ''; -- 10. Display all members SELECT id, full_name, referral_code, referred_by FROM public.profiles ORDER BY full_name;
